# ── Step 1: Extract all zips ─────────────────────────────────────────────────

cd data/Squat    && unzip videos.zip -d videos/
cd ../OHP        && unzip videos.zip -d videos/
cd ../BarbellRow && unzip BarbellRow_images.zip -d images/
cd ../Squat/Labeled_Dataset/labels/Shallow_Squat_Error_Dataset && unzip *.zip -d images/
cd ../../../../..   # back to project root


# ── Step 2: Dry run — verify all paths load cleanly (no pose extraction) ─────

python train.py --exercise Squat      --dry_run
python train.py --exercise OHP        --dry_run
python train.py --exercise BarbellRow --dry_run


# ── Step 3: Sanity check — small cap, runs in ~5 min each ────────────────────

python train.py --exercise Squat      --max_samples 100
python train.py --exercise OHP        --max_samples 100
python train.py --exercise BarbellRow --max_samples 300


# ── Step 4: Full training — all three exercises ───────────────────────────────

python train.py --exercise all


# ── Or train each individually if you want to monitor separately ──────────────

python train.py --exercise Squat

python train.py --exercise BarbellRow

ls -lh models/
